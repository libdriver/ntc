51{
52 uint8_t res;

54 /* link interface function */
55 DRIVER_NTC_LINK_INIT(&gs_handle, ntc_handle_t);
56 DRIVER_NTC_LINK_ADC_INIT(&gs_handle, ntc_interface_adc_init);
57 DRIVER_NTC_LINK_ADC_DEINIT(&gs_handle, ntc_interface_adc_deinit);
58 DRIVER_NTC_LINK_ADC_READ(&gs_handle, ntc_interface_adc_read);
59 DRIVER_NTC_LINK_DELAY_MS(&gs_handle, ntc_interface_delay_ms);
60 DRIVER_NTC_LINK_DEBUG_PRINT(&gs_handle, ntc_interface_debug_print);
61
62 /* init */
63 res = ntc_init(&gs_handle);
64 if (res != 0)
65 {
66 ntc_interface_debug_print("ntc: init failed.\n");
67
68 return 1;
69 }
70
71 /* set circuit */
72 res = ntc_set_circuit(&gs_handle, circuit);
73 if (res != 0)
74 {
75 ntc_interface_debug_print("ntc: set circuit.\n");
76 (void)ntc_deinit(&gs_handle);
77
78 return 1;
79 }
80
81 /* set circuit fixed resistor */
82 res = ntc_set_circuit_fixed_resistor(&gs_handle, r_fixed_ohm);
83 if (res != 0)
84 {
85 ntc_interface_debug_print("ntc: set circuit fixed resistor.\n");
86 (void)ntc_deinit(&gs_handle);
87
88 return 1;
89 }
90
91 return 0;
92}
